async function loadData() { try { let response = await fetch('./data.json'); let rawData = await response.json(); // แปลงข้อมูลจาก Google Sheet ให้เข้ากับตารางแดชบอร์ด database = { 'โรงไฟฟ้า 34': { 'รวมเชื้อเพลิง': [] } }; rawData.forEach(row => { let netCv = parseFloat(row['Net CV ARB (kcal/kg)']) || 0; let price = parseFloat(row['Price']) || 0; let moisture = parseFloat(row['%TM'] || row['ความชื้นจริง(ชิพ)']) || 0; let ton = parseFloat(row['นน.สุทธิ']) || 0; let bahtMj = netCv > 0 ? parseFloat((price / netCv).toFixed(4)) : 0; database['โรงไฟฟ้า 34']['รวมเชื้อเพลิง'].push({ tier: 'B', name: row['ชื่อลูกค้า'] || 'ไม่ระบุชื่อ', ton: ton, moisture: moisture, netCv: netCv, price: price, bahtMj: bahtMj, variance: 0, alert: false, alertText: '', history: [bahtMj * 0.98, bahtMj * 1.01, bahtMj] }); }); switchPlant('โรงไฟฟ้า 34'); } catch (error) { console.error("ไม่สามารถโหลดไฟล์ data.json ได้:", error); document.getElementById('aiRecommendationText').innerHTML = "⚠️ ไม่พบไฟล์ข้อมูล ./data.json กรุณารัน GitHub Action ครั้งแรกก่อน"; } }
