async function loadData() { try { let response = await fetch('./data.json'); let rawData = await response.json(); database = {}; rawData.forEach(row => { // วนหาคีย์ที่ชื่อคล้ายคำว่า "โรงไฟฟ้า" เพื่อป้องกันปัญหาเว้นวรรคใน Google Sheet let plantKey = Object.keys(row).find(k => k.trim() === 'โรงไฟฟ้า') || 'โรงไฟฟ้า'; let fuelKey = Object.keys(row).find(k => k.trim() === 'สินค้า') || 'สินค้า'; let plant = (row[plantKey] || 'ไม่ระบุโรงไฟฟ้า').trim(); let fuel = (row[fuelKey] || 'รวมเชื้อเพลิง').trim(); let netCv = parseFloat(row['Net CV ARB (kcal/kg)'] || row['Net CV (kcal/kg)']) || 0; let price = parseFloat(row['Price'] || row['ราคา/ตัน']) || 0; let moisture = parseFloat(row['%TM'] || row['ความชื้นจริง(ชิพ)']) || 0; let ton = parseFloat(row['นน.สุทธิ']) || 0; let bahtMj = netCv > 0 ? parseFloat((price / netCv).toFixed(4)) : 0; if (!database[plant]) database[plant] = {}; if (!database[plant][fuel]) database[plant][fuel] = []; database[plant][fuel].push({ tier: 'B', name: row['ชื่อลูกค้า'] || 'ไม่ระบุชื่อ', ton: ton, moisture: moisture, netCv: netCv, price: price, bahtMj: bahtMj, variance: 0, alert: false, alertText: '', history: [bahtMj * 0.98, bahtMj * 1.01, bahtMj] }); }); const plantTabsContainer = document.getElementById('plantTabs'); plantTabsContainer.innerHTML = ''; const plants = Object.keys(database); if (plants.length > 0) { currentPlant = plants[0]; plants.forEach(p => { let btn = document.createElement('button'); btn.className = `plant-tab ${p === currentPlant ? 'active' : ''}`; btn.textContent = `⚡ ${p}`; btn.onclick = () => switchPlant(p); plantTabsContainer.appendChild(btn); }); switchPlant(currentPlant); } else { document.getElementById('aiRecommendationText').innerHTML = "⚠️ ไม่พบข้อมูลโรงไฟฟ้าในไฟล์ data.json"; } } catch (error) { console.error("ไม่สามารถโหลดไฟล์ data.json ได้:", error); document.getElementById('aiRecommendationText').innerHTML = "⚠️ ไม่พบไฟล์ข้อมูล ./data.json กรุณารัน GitHub Action ก่อน"; } }
